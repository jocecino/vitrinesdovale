-- ===================================================================== -- Vitrines do Vale: avaliações + "vendidos" -- Cole TUDO no Supabase > SQL Editor > Run (pode rodar mais de uma vez). -- -- Suposições sobre o seu banco (confira): -- * tabela public.pedidos com colunas: numero, tel, status e itens (jsonb) -- onde cada item tem o id do produto em "id" (ou "produto_id") e a -- quantidade em "qtd" (ou "q"). -- * tabela public.produtos com coluna loja_id. -- * já existe a função consultar_pedidos(p_lista) usada pelo app. -- Se "vendidos" aparecer 0 mesmo com pedidos entregues, me envie o resultado de: -- select column_name, data_type from information_schema.columns -- where table_name in ('pedidos','produtos'); -- ===================================================================== create table if not exists public.avaliacoes ( id uuid primary key default gen_random_uuid(), pedido_num bigint not null, produto_id text not null, loja_id text, nome text, nota int not null check (nota between 1 and 5), texto text, foto text, visivel boolean not null default true, criado_em timestamptz not null default now(), unique (pedido_num, produto_id) ); alter table public.avaliacoes enable row level security; -- sem policies: o acesso é só pelas funções abaixo. -- Pasta pública para as fotos que os clientes enviam insert into storage.buckets (id, name, public) values ('avaliacoes','avaliacoes',true) on conflict (id) do nothing; drop policy if exists "avaliacoes_upload" on storage.objects; create policy "avaliacoes_upload" on storage.objects for insert to anon, authenticated with check (bucket_id = 'avaliacoes' and (storage.foldername(name))[1] = 'a'); drop policy if exists "avaliacoes_ler" on storage.objects; create policy "avaliacoes_ler" on storage.objects for select using (bucket_id = 'avaliacoes'); -- --------------------------------------------------------------------- -- auxiliar: acha o pedido pelo número + telefone (últimos 9 dígitos) -- --------------------------------------------------------------------- create or replace function public._pedido_do_cliente(p_numero bigint, p_tel text) returns jsonb language sql stable security definer set search_path = public as $$ select to_jsonb(x) from public.pedidos x where (to_jsonb(x)->>'numero')::bigint = p_numero and right(regexp_replace(coalesce(to_jsonb(x)->>'tel',''),'\D','','g'),9) = right(regexp_replace(coalesce(p_tel,''),'\D','','g'),9) and length(regexp_replace(coalesce(p_tel,''),'\D','','g')) >= 9 limit 1; $$; revoke all on function public._pedido_do_cliente(bigint,text) from public, anon, authenticated; -- --------------------------------------------------------------------- -- itens do pedido que podem ser avaliados (só pedidos ENTREGUES) -- --------------------------------------------------------------------- create or replace function public.itens_avaliar(p_numero bigint, p_tel text) returns jsonb language plpgsql stable security definer set search_path = public as $$ declare ped jsonb; r jsonb; begin ped := public._pedido_do_cliente(p_numero, p_tel); if ped is null or ped->>'status' <> 'entregue' then return '[]'::jsonb; end if; select coalesce(jsonb_agg(jsonb_build_object( 'id', coalesce(it->>'id', it->>'produto_id'), 't', coalesce(it->>'titulo', it->>'t', 'Produto'))), '[]'::jsonb) into r from jsonb_array_elements(coalesce(ped->'itens','[]'::jsonb)) it where coalesce(it->>'id', it->>'produto_id') is not null; return r; end $$; grant execute on function public.itens_avaliar(bigint,text) to anon, authenticated; -- --------------------------------------------------------------------- -- enviar avaliação p = {numero, tel, nome, itens:[{id,nota,texto,foto}]} -- --------------------------------------------------------------------- create or replace function public.avaliar(p jsonb) returns boolean language plpgsql security definer set search_path = public as $$ declare ped jsonb; it jsonb; pid text; nm text; ok_ids text[]; lj text; n int; begin ped := public._pedido_do_cliente((p->>'numero')::bigint, p->>'tel'); if ped is null then raise exception 'Pedido não encontrado.'; end if; if ped->>'status' <> 'entregue' then raise exception 'Você só pode avaliar depois que o pedido for entregue.'; end if; select array_agg(coalesce(i->>'id', i->>'produto_id')) into ok_ids from jsonb_array_elements(coalesce(ped->'itens','[]'::jsonb)) i; -- nome exibido: primeiro nome + inicial do último (privacidade) nm := trim(coalesce(nullif(p->>'nome',''), ped->>'cliente', ped->>'nome', 'Cliente')); if position(' ' in nm) > 0 then nm := split_part(nm,' ',1) || ' ' || left(regexp_replace(nm,'^.*\s',''),1) || '.'; end if; nm := left(nm, 40); for it in select * from jsonb_array_elements(coalesce(p->'itens','[]'::jsonb)) loop pid := it->>'id'; if pid is null or not (pid = any(ok_ids)) then continue; end if; n := (it->>'nota')::int; if n is null or n < 1 or n > 5 then raise exception 'Nota inválida.'; end if; select (to_jsonb(pr)->>'loja_id') into lj from public.produtos pr where pr.id::text = pid; insert into public.avaliacoes(pedido_num, produto_id, loja_id, nome, nota, texto, foto) values ((p->>'numero')::bigint, pid, lj, nm, n, left(coalesce(it->>'texto',''),500), nullif(it->>'foto','')); end loop; return true; exception when unique_violation then raise exception 'Este pedido já foi avaliado. Obrigado!'; end $$; grant execute on function public.avaliar(jsonb) to anon, authenticated; -- --------------------------------------------------------------------- -- dados sociais do produto: vendidos (entregues), média, total e lista -- --------------------------------------------------------------------- create or replace function public.produto_social(p_id text) returns jsonb language plpgsql stable security definer set search_path = public as $$ declare v bigint; m numeric; t int; l jsonb; begin select coalesce(sum(coalesce((it->>'qtd')::int,(it->>'q')::int,1)),0) into v from public.pedidos x, jsonb_array_elements(coalesce(to_jsonb(x)->'itens','[]'::jsonb)) it where to_jsonb(x)->>'status' = 'entregue' and coalesce(it->>'id', it->>'produto_id') = p_id; select count(*), coalesce(round(avg(nota)::numeric,1),0) into t, m from public.avaliacoes where produto_id = p_id and visivel; select coalesce(jsonb_agg(jsonb_build_object( 'nome',nome,'nota',nota,'texto',texto,'foto',foto, 'data',to_char(criado_em at time zone 'America/Bahia','DD/MM/YYYY')) order by criado_em desc), '[]'::jsonb) into l from (select * from public.avaliacoes where produto_id = p_id and visivel order by criado_em desc limit 30) a; return jsonb_build_object('vendidos',v,'media',m,'total',t,'lista',l); end $$; grant execute on function public.produto_social(text) to anon, authenticated;
